      if (!db) { cargarDatosLocalStorage(); renderizarTodo(); return; }
      const transaction = db.transaction(["stateStore"], "readonly");
      const store = transaction.objectStore("stateStore");
      const request = store.get("currentState");
      request.onsuccess = function(event) {
        if (event.target.result) state = event.target.result;
        else cargarDatosLocalStorage();
        renderizarTodo();
      };
      request.onerror = function() { cargarDatosLocalStorage(); renderizarTodo(); };
    }

    function actualizarMetricasSuperiores() {
      let liquidoTotal = state.tarjetas.filter(t => t.tipo === 'debito').reduce((acc, t) => acc + (t.saldoDebito || 0), 0);
      document.getElementById('metricLiquidoTotal').innerHTML = `$${liquidoTotal.toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2})} <span class="text-xs font-normal text-slate-400">MXN</span>`;

      let deudaGlobalTotal = 0;
      let limiteGlobalTotal = 0;
      state.tarjetas.filter(t => t.tipo === 'credito').forEach(t => {
        let abonosTarjeta = state.abonos.filter(a => a.tarjetaId === t.id).reduce((sum, a) => sum + a.monto, 0);
        let pendiente = Math.max(0, (t.deuda || 0) - abonosTarjeta);
        deudaGlobalTotal += pendiente;
        limiteGlobalTotal += (t.limite || 0);
      });

      document.getElementById('metricPendienteTotal').innerHTML = `$${deudaGlobalTotal.toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2})} <span class="text-xs font-normal text-slate-400">MXN</span>`;

      let balanceNeto = liquidoTotal - deudaGlobalTotal;
      const badgeBalanceNeto = document.getElementById('badgeBalanceNeto');
      document.getElementById('metricBalanceNeto').innerHTML = `$${balanceNeto.toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2})} <span class="text-xs font-normal text-slate-400">MXN</span>`;
      
      if (balanceNeto >= 0) {
        badgeBalanceNeto.className = "text-xs bg-emerald-100 text-emerald-800 px-2 py-0.5 rounded-md font-bold";
        badgeBalanceNeto.innerText = "Solvente (+)";
      } else {
        badgeBalanceNeto.className = "text-xs bg-rose-100 text-rose-800 px-2 py-0.5 rounded-md font-bold";
        badgeBalanceNeto.innerText = "Déficit (-)";
      }

      let usoPorcentaje = limiteGlobalTotal > 0 ? (deudaGlobalTotal / limiteGlobalTotal) * 100 : 0;
      document.getElementById('metricUsoCredito').innerHTML = `${usoPorcentaje.toFixed(1)}% <span class="text-xs font-normal text-slate-400">de $${(limiteGlobalTotal/1000).toFixed(0)}k</span>`;

        badgeSalud.innerText = "Moderado";
      }
    }

    function evaluarTarjeta() {
      const fechaCompraVal = document.getElementById('fechaCompra').value;
      const categoria = document.getElementById('categoriaGasto').value;
      const monto = parseFloat(document.getElementById('montoGasto').value) || 0;
      const fechaCompra = new Date(fechaCompraVal || Date.now());
      const tarjetasActivas = state.tarjetas.filter(t => t.activa);

      if (tarjetasActivas.length === 0) return;

      let mejorTarjeta = null;
      let maxPuntuacion = -9999;
      let razonMejor = "";
      let diasFinanciamientoCalc = 0;
      let beneficioEstimado = 0;

      tarjetasActivas.forEach(t => {
        let puntuacion = 0;
        let razon = "";
        let diasFin = 0;
        let beneficio = 0;

        if (t.tipo === 'debito') {
          puntuacion += 10;
          razon = `Cuenta débito (${t.nombre}). Gasta sin comisiones ni endeudamiento.`;
          beneficio = (monto * (t.tasaAnual || 0) / 100) / 12;
        } else {
          let corte = t.corte;
          let pago = t.pago;
          let proximoCorte = new Date(fechaCompra.getFullYear(), fechaCompra.getMonth(), corte);
          if (fechaCompra > proximoCorte) proximoCorte.setMonth(proximoCorte.getMonth() + 1);
          let proximoPago = new Date(proximoCorte.getFullYear(), proximoCorte.getMonth(), pago);
          if (proximoPago <= proximoCorte) proximoPago.setMonth(proximoPago.getMonth() + 1);

          diasFin = Math.ceil(Math.abs(proximoPago - fechaCompra) / (1000 * 60 * 60 * 24));
          puntuacion += diasFin * 1.5;
          razon = `Compra post-corte (${corte}). Aprox. ${diasFin} días de financiamiento.`;

          if (t.nombre.toLowerCase().includes('santander') || t.nombre.toLowerCase().includes('likeu')) {
            if (categoria === 'gasolina') { beneficio = monto * 0.04; puntuacion += 50; razon += ` ¡Cashback 4% Gasolina!`; }
            else if (categoria === 'farmacia') { beneficio = monto * 0.06; puntuacion += 60; razon += ` ¡Cashback 6% Farmacia!`; }
          }
        }

        if (puntuacion > maxPuntuacion) {
          maxPuntuacion = puntuacion;
          mejorTarjeta = t;
          razonMejor = razon;
          diasFinanciamientoCalc = t.tipo === 'credito' ? diasFin : 30;
          beneficioEstimado = beneficio;
        }
      });

      if (mejorTarjeta) {
        document.getElementById('tarjetaNombre').innerText = mejorTarjeta.nombre;
        document.getElementById('tarjetaRazon').innerText = razonMejor;
        document.getElementById('badgeFinanciamiento').innerText = mejorTarjeta.tipo === 'credito' ? `${diasFinanciamientoCalc} días financiamiento` : `Pago Débito`;
        document.getElementById('tarjetaCorte').innerText = mejorTarjeta.tipo === 'credito' ? `Día ${mejorTarjeta.corte}` : `N/A`;
        document.getElementById('tarjetaPago').innerText = mejorTarjeta.tipo === 'credito' ? `Día ${mejorTarjeta.pago}` : `Inmediato`;
        document.getElementById('rendimientoEstimadoText').innerText = `$${beneficioEstimado.toFixed(2)} MXN`;
      }
    }

    function initChartNu() {
      const ctx = document.getElementById('chartNu').getContext('2d');
      let baseLiquida = state.tarjetas.filter(t => t.tipo === 'debito').reduce((sum, t) => sum + (t.saldoDebito || 25000), 0);
      let tasaMensual = 0.135 / 12;
      let meses = ['Actual', 'Mes 1', 'Mes 2', 'Mes 3', 'Mes 4', 'Mes 5', 'Mes 6'];
      let dataPuntos = [baseLiquida];
      let currentVal = baseLiquida;

      for (let i = 1; i <= 6; i++) {
        currentVal += currentVal * tasaMensual;
        dataPuntos.push(Math.round(currentVal));
      }

      if (chartNuInstance) chartNuInstance.destroy();
      chartNuInstance = new Chart(ctx, {
        type: 'line',
        data: {
          labels: meses,
          datasets: [{ data: dataPuntos, borderColor: '#6366f1', backgroundColor: 'rgba(99, 102, 241, 0.1)', borderWidth: 3, fill: true, tension: 0.35 }]
        },
        options: { responsive: true, maintainAspectRatio: false, plugins: { legend: { display: false } } }
      });
    }

    function calcularMSI() {
      const monto = parseFloat(document.getElementById('msiMonto').value) || 0;
      const plazo = parseInt(document.getElementById('msiPlazo').value) || 12;
      const tasaAnual = parseFloat(document.getElementById('msiTasa').value) || 13.5;
      const pagoMensual = monto / plazo;
      const tasaMensual = (tasaAnual / 100) / 12;

      let saldoNu = monto;
      let rendimientoTotal = 0;
      const tbody = document.getElementById('tablaMSIBody');
      tbody.innerHTML = "";

      for (let m = 1; m <= plazo; m++) {
        let rendMes = saldoNu * tasaMensual;
        rendimientoTotal += rendMes;
        saldoNu += rendMes;
        let saldoAntes = saldoNu;
        saldoNu -= pagoMensual;

        if (m <= 4 || m === plazo) {
          const tr = document.createElement('tr');
          tr.className = "hover:bg-slate-50 transition border-b border-slate-100";
          tr.innerHTML = `
            <td class="p-3.5 font-bold text-slate-900">Mes ${m}</td>
            <td class="p-3.5 text-slate-700">$${saldoAntes.toLocaleString('en-US', {maximumFractionDigits: 2})}</td>
            <td class="p-3.5 font-bold text-emerald-600">+$${rendMes.toLocaleString('en-US', {maximumFractionDigits: 2})}</td>
            <td class="p-3.5 text-rose-600 font-bold">-$${pagoMensual.toLocaleString('en-US', {maximumFractionDigits: 2})}</td>
            <td class="p-3.5 font-black text-slate-900">$${Math.max(0, saldoNu).toLocaleString('en-US', {maximumFractionDigits: 2})}</td>
          `;
          tbody.appendChild(tr);
        }
      }

      let descuento = monto > 0 ? (rendimientoTotal / monto) * 100 : 0;
      document.getElementById('msiPagoMensual').innerText = `$${pagoMensual.toLocaleString('en-US', {minimumFractionDigits: 2})} MXN`;
      document.getElementById('msiRendimientoTotal').innerText = `+$${rendimientoTotal.toLocaleString('en-US', {minimumFractionDigits: 2})} MXN`;
      document.getElementById('msiDescuentoEfectivo').innerText = `${descuento.toFixed(2)}%`;
    }

    function guardarNuevaTarjeta() {
      const nombre = document.getElementById('nuevaTarjetaNombre').value.trim();
      const tipo = document.getElementById('nuevaTarjetaTipo').value;
      if (!nombre) { mostrarToast("Ingresa un nombre.", "⚠️"); return; }

      let nueva = { id: Date.now(), nombre, tipo, activa: true, color: tipo === 'debito' ? 'bg-emerald-600' : 'bg-indigo-600' };
      if (tipo === 'debito') {
        nueva.saldoDebito = parseFloat(document.getElementById('nuevaTarjetaSaldoDebito').value) || 0;
        nueva.tasaAnual = parseFloat(document.getElementById('nuevaTarjetaTasa').value) || 0;
      } else {
        nueva.corte = parseInt(document.getElementById('nuevaTarjetaCorte').value) || 1;
        nueva.pago = parseInt(document.getElementById('nuevaTarjetaPago').value) || 15;
        nueva.limite = parseFloat(document.getElementById('nuevaTarjetaLimite').value) || 20000;
        nueva.deuda = parseFloat(document.getElementById('nuevaTarjetaDeuda').value) || 0;
      }

      state.tarjetas.push(nueva);
      toggleModalNuevaTarjeta(false);
      renderizarTodo();
      mostrarToast(`Tarjeta '${nombre}' guardada.`, "💳");
    }

    function guardarGastoManual() {
      const comercio = document.getElementById('gastoComercio').value.trim();
      const monto = parseFloat(document.getElementById('gastoMonto').value) || 0;
      const fecha = document.getElementById('gastoFecha').value;
      const categoria = document.getElementById('gastoCategoria').value;
      const tarjetaId = parseInt(document.getElementById('gastoTarjetaSelect').value);
      if (!comercio || monto <= 0) { mostrarToast("Completa los campos.", "⚠️"); return; }

      let cashback = 0;
      const tarjeta = state.tarjetas.find(t => t.id === tarjetaId);
      if (tarjeta && tarjeta.tipo === 'credito') {
        if (categoria === 'gasolina') cashback = monto * 0.04;
        if (categoria === 'farmacia') cashback = monto * 0.06;
      } else if (tarjeta && tarjeta.tipo === 'debito') {
        tarjeta.saldoDebito = Math.max(0, (tarjeta.saldoDebito || 0) - monto);
      }

      state.gastos.push({ id: Date.now(), fecha, comercio, categoria, tarjetaId, monto, cashback });
      toggleModalGasto(false);
      renderizarTodo();
      switchTab('tab3');
      mostrarToast("Gasto registrado.", "📋");
    }
